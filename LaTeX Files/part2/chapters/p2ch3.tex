% --- 3.1 بناء المخطط الصندوقي ---
\subsection{بناء المخطط الصندوقي \el{(Circuit Topology)}}

لتحقيق استجابة "تمرير منخفض"، تم إدراج ملفات خنق \el{(Inductors)} على التوالي مع خطوط التغذية الثلاثة الواصلة إلى المقوم. بعد هذه الملفات مباشرة، تم توصيل فروع مكونة من مقاومة ومكثف على التوالي موصلة نجمياً \el{(Wye-connected)} إلى الأرضي. تضمن هذه البنية إجبار التيارات التوافقية ذات التردد العالي على المرور عبر المسار التفرعي إلى الأرضي، بينما يمر التردد الأساسي \el{($50 Hz$)} إلى الحمل.

% --- 3.2 إدخال بارامترات العناصر ---
\subsection{إدخال بارامترات العناصر}

تم ضبط قيم العناصر بناءً على الحسابات النظرية التي أجريناها سابقاً لتردد رنيني قدره \el{$150 Hz$}. توضح الأشكال التالية كيفية إدخال هذه القيم داخل بلوكات \el{Simulink}:

\begin{enumerate}
    \item \textbf{الملفات التسلسلية \el{(Inductances)}:} تم استخدام بلوك \el{\textit{Series RLC Branch}} وضبط نوع الفرع على $L$ فقط، مع إدخال القيمة المحسوبة $0.002\ H$ (أو $2\ mH$) لضمان توفير الممانعة اللازمة عند الترددات العالية.
    
    \begin{figure}[H]
        \centering
        \includegraphics[width=0.85\linewidth]{15 - Inductances of Low-Pass filter.png}
        \caption{نافذة بارامترات ملفات الخنق في النموذج المطور}
    \end{figure}

    \item \textbf{مقاومات التخميد \el{(Resistances)}:}في الفروع التفرعية، تم ضبط نوع الفرع على $R$ وإدخال القيمة $0.628 \Omega$. تعمل هذه المقاومة على تخميد الرنين وضمان استقرار النظام ومنع التذبذبات غير المرغوبة.
    \begin{figure}[H]
        \centering
        \includegraphics[width=0.85\linewidth]{16 - Resistances of Low-Pass filter.png}
        \caption{إدخال قيم المقاومات الخاصة بفرع الترشيح}
    \end{figure}

    \clearpage

    \item \textbf{المكثفات \el{(Capacitances)}:}تم ضبط نوع الفرع التفرعي ليتضمن السعة، وإدخال القيمة الدقيقة $562.9 \times 10^{-6} F$ (أو $562.9 \mu F$).
    \begin{figure}[H]
        \centering
        \includegraphics[width=0.85\linewidth]{17 - Capacitances of Low-Pass filter.png}
        \caption{ضبط قيم السعات لمرشح التمرير المنخفض}
    \end{figure}
\end{enumerate}

\clearpage
