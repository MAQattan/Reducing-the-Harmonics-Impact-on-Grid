\section{توصيف المنظومة والنمذجة} \label{part1:sec1}
    % \thispagestyle{empty} % This removes header/footer from the current page only
    {\fontsize{16}{27}\selectfont
    تتمثل الخطوة الأساسية في هذه الدراسة في بناء نموذج دقيق للنظام الكهربائي باستخدام مكتبة \el{Specialized Power Systems} في بيئة \el{MATLAB R2023b}. تم تصميم الدارة لتمثيل شبكة تغذية تغذي حملاً غير خطي، مع الأخذ بعين الاعتبار الممانعات المختلفة التي تؤثر على جودة القدرة.
    \par}

\subsection{مكونات الشبكة \el{(Network Components)}}
يتألف النموذج من العناصر التالية، والتي تم ضبط بارامتراتها بناءً على الحسابات الهندسية الموضحة في الجلسة الثانية:
\begin{enumerate}
  \item \textbf{منبع الجهد \el{(Three-Phase Source)}}:
  يمثل محطة التحويل بجهد اسمي قدره $400 V$ (بين الأطوار) وتردد $50 Hz$. تم ضبط قيمة الذروة في الموديل لتكون $220\sqrt{2} V$.\\[-10pt]
  
\begin{arabicbox}{تذكر}
  يجب ضبط الإزاحة الطورية بين المنابع الثلاثة بزاوية $120^\circ$ لتكون ($0^\circ, -120^\circ, 120^\circ$) للأطوار \el{A} و \el{B} و \el{C} على التوالي، لتجنب ظهور تيارات تسلسل صفري غير مرغوبة.
\end{arabicbox}
 

  \item \textbf{ممانعة القصر للشبكة ($Z_{sc}$)}: تم تمثيلها باستخدام بلوك \el{Series RLC Branch} بقيم محسوبة لتعكس استطاعة قصر قدرها $500 MVA$، حيث بلغت المقاومة $R_{sc} = 3.184 \times 10^{-3} \Omega$ والحث $L_{sc} = 1 mH$ لضمان ظهور تشوه الجهد بوضوح عند التحليل.
  \item \textbf{ممانعة المحولة ($Z_{T}$)}:
  تم تمثيل المحولة ($1000 KVA$) بممانعة تسلسلية منسوبة للطرف الثانوي قيمتها $R_T = 2.08 \times 10^{-3} \Omega$ و $L_T = 3.05 \times 10^{-5} H$.
  \item \textbf{ممانعة الكبل \el{(Cable Impedance)}}: كبل نحاسي بطول $230 m$ ومقطع $16 mm^2$، بممانعة قدرها $R = 0.256 \Omega$ و $L = 6.05 \times 10^{-5} H$.
  \item \textbf{الحمل غير الخطي \el{(Non-linear Load)}}:
  يتكون من جسر تقويم ديودي سداسي النبضات \el{(Universal Bridge)} يغذي حملاً مستمراً \el{(RL Load)} باستطاعة $25 KW$، حيث تبلغ مقاومة الحمل $10.6 \Omega$ وتحريضه $1 mH$ .
\end{enumerate}

\subsection{إعدادات المحاكاة \el{(Simulation Settings)}}

للحصول على نتائج دقيقة في تحليل التوافقيات \el{($FFT Analysis$)}، تم ضبط بلوك \el{powergui} وفق المعايير التالية:
\begin{itemize}
    \item تم اختيار \textbf{\el{Simulation type}} على الوضع \el{Discrete} (متقطع) بدلاً من الوضع المستمر، وذلك لأن النظام يحتوي على عناصر إلكترونيات قدرة \el{(Diodes)} تتطلب تمثيلاً دقيقاً لعمليات التبديل.
    \item تم تحديد زمن العينة  \textbf{\el{Sample time}} بقيمة دقيقة جداً وهي $1 \times 10^{-6} s$. هذا الزمن الصغير ضروري جداً لالتقاط الرتب التوافقية العالية بدقة ومنع حدوث أخطاء "الطي" \el{(Aliasing)} في الطيف الترددي.
\end{itemize}

\section{الحسابات النظرية\\ للمرشحات} \label{part1:sec2}
%   \thispagestyle{empty} % This removes header/footer from the current page only
  {\fontsize{16}{27}\selectfont
  لتحسين جودة القدرة وتخفيض نسبة التشوه التوافقي ($THD$)، يتم تصميم مرشحات تردد رنيني \el{\textbf{(Tuned Filters)}} تسلسلية من النوع ($RLC$). الهدف من هذه المرشحات هو توفير مسار منخفض الممانعة لتيارات التوافقيات عند ترددات الرنين الخاصة بها، مما يمنعها من الانتشار في الشبكة.
  \par}
 \vspace{5pt} 
{\sloppy
\noindent تعتمد الحسابات على البارامترات الأساسية التالية المستخرجة من "الجلسة الثالثة":}
\begin{itemize}
    \item \textbf{التردد الأساسي ($f$):} $50 Hz$.
    \item \textbf{سعة المكثف ($C$):} تم حسابها سابقاً لتصحيح عامل الاستطاعة وهي ثابتة لكافة المرشحات: $C = 3.339 \times 10^{-5} F$.
    \item \textbf{عامل الجودة ($Q$):} تم اختياره بقيمة $75$ لتحقيق دقة ترشيح مناسبة وتوازن بين عرض الحزمة وممانعة الرنين.
\end{itemize}

\subsection{قوانين التصميم المستخدمة}
تعتمد عملية الحساب على القوانين الفيزيائية التالية:
\begin{enumerate}
    \item \textbf{تردد الرنين ($f_r$):} $f_r = n \times f$ (حيث $n$ هي رتبة التوافقية).
    \item \textbf{قانون الحث ($L$):} يُشتق من علاقة الرنين الأساسية $f_r = \frac{1}{2 \pi \sqrt{LC}}$ ليصبح:
    \begin{equation*}
        L_n = \frac{1}{(2 \pi f_r)^2 \times C}
    \end{equation*}
    \item \textbf{قانون المقاومة ($r$):} يُشتق من علاقة عامل الجودة $Q = \frac{X_0}{r}$ حيث $X_0 = \sqrt{\frac{L}{C}}$، ليصبح:
    \begin{equation*}
        r_n = \frac{\sqrt{L_n / C}}{Q}
    \end{equation*}
\end{enumerate}

\subsection{حسابات مرشح التوافقية السابعة ($n=7$)}
\begin{itemize}
    \item \textbf{تردد الرنين:} $f_{r7} = 7 \times 50 = 350 Hz$.
    \item \textbf{حساب الحث ($L_7$):}
    \begin{equation*}
        L_7 = \frac{1}{(2 \pi \times 350)^2 \times 3.339 \times 10^{-5}} \approx \mathbf{0.00619 H} \text{ أو } (6.19 mH)
    \end{equation*}
    \item \textbf{حساب المقاومة ($r_7$):}
    \begin{equation*}
        r_7 = \frac{\sqrt{0.00619 / 3.339 \times 10^{-5}}}{75} \approx \mathbf{0.1815 \Omega}
    \end{equation*}
\end{itemize}

\subsection{حسابات مرشح التوافقية الحادية عشرة ($n=11$)}
\begin{itemize}
    \item \textbf{تردد الرنين:} $f_{r11} = 11 \times 50 = 550 Hz$.
    \item \textbf{حساب الحث ($L_{11}$):}
    \begin{equation*}
        L_{11} = \frac{1}{(2 \pi \times 550)^2 \times 3.339 \times 10^{-5}} \approx \mathbf{0.0025 H} \text{ أو } (2.5 mH)
    \end{equation*}
    \item \textbf{حساب المقاومة ($r_{11}$):}
    \begin{equation*}
        r_{11} = \frac{\sqrt{0.0025 / 3.339 \times 10^{-5}}}{75} \approx \mathbf{0.115 \Omega}
    \end{equation*}
\end{itemize}

\subsection{إعداد أدوات القياس}
لتمكين البرنامج من تحليل الإشارات، تم إجراء تعديلات تقنية هامة على كتل السكوب \el{(Scopes)} الخاصة بالجهد والتيار قبل المقوم:
\begin{itemize}
    \item \textbf{تسجيل البيانات \el{(Data Logging)}:} تم تفعيل خيار \el{\textbf{Log data to workspace}} لضمان تصدير قيم الجهد والتيار إلى مساحة عمل ماتلاب.
    \item \textbf{تنسيق البيانات \el{(Save Format)}:} تم ضبط تنسيق الحفظ على \el{\textbf{Structure with time}}، وهو التنسيق الوحيد الذي تتعرف عليه أداة تحليل \el{FFT} في النسخ الحديثة.
    \item \textbf{إعدادات الموديل:} تم إلغاء تحديد خيار \el{\textbf{Single simulation output}} من إعدادات الموديل \el{(Model Settings)} لضمان ظهور المتغيرات بشكل مستقل في قائمة \el{FFT Analyzer}.
    \item \textbf{اسم المتحولة \el{(Variable Name)}:} تم تسمية المتحولة الخاصة بالتيار \el{"CurrentData"} وبالجهد \el{"VoltageData"} لتسهيل التعرف عليها أثناء التحليل.
\end{itemize}

\section{مناقشة النتائج} \label{part1:sec5}
% \thispagestyle{empty}

من خلال هذه الدراسة والمحاكاة العملية باستخدام برنامج \el{\textbf{MATLAB R2023b}}، خرجنا بالنتائج التالية:

\begin{enumerate}
    \item أثبتت مرشحات التردد الرنيني \el{(\textbf{Tuned Filters})} كفاءة عالية في امتصاص تيارات التوافقيات عند ترددات الرنين المحددة، مما أدى لخفض نسبة \el{$THD$} بشكل تدريجي مع كل إضافة.
    \item ساهمت المرشحات في تحسين معامل الاستطاعة وتقريب شكل موجة التيار والجهد من الشكل الجيبي النقي، مما يقلل من الفواقد والحرارة في عناصر الشبكة.
    \item بفضل إضافة مرشحات الرتب 7 و 11، تمكنا من الحفاظ على جودة الجهد ضمن الحدود العالمية المسموح بها (غالباً أقل من 3\% أو 5\% حسب المواصفة).
    \item أكدت الدراسة أن للحصول على نتائج صحيحة ومطابقة للواقع الفيزيائي يجب أن ضبط إعدادات المحاكاة بشكل دقيق و إدخال بارامترات العناصر بعناية شديدة، حيث أن أي خطأ في القيم أو الإعدادات قد يؤدي لنتائج غير دقيقة أو غير منطقية.
\end{enumerate}
